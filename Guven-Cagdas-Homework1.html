import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from spatialmath import SE3
from matplotlib.animation import FuncAnimation

# Define cube vertices and edges
def cube_vertices_edges():
    # Cube vertices centered at origin
    half_size = 0.5
    vertices = np.array([
        [ half_size,  half_size,  half_size],
        [ half_size,  half_size, -half_size],
        [ half_size, -half_size,  half_size],
        [ half_size, -half_size, -half_size],
        [-half_size,  half_size,  half_size],
        [-half_size,  half_size, -half_size],
        [-half_size, -half_size,  half_size],
        [-half_size, -half_size, -half_size]
    ])
    
    # Define edges by pairs of vertices
    edges = [
        [0, 1], [0, 2], [0, 4], [1, 3], [1, 5],
        [2, 3], [2, 6], [3, 7], [4, 5], [4, 6],
        [5, 7], [6, 7]
    ]
    
    return vertices, edges

# Plot the cube with transformation applied
def plot_cube(ax, vertices, edges, transform=None):
    transformed_vertices = []
    
    if transform is not None:
        # Apply the transformation to each vertex
        transformed_vertices = np.array([transform @ np.append(v, 1) for v in vertices])[:, :3]
    else:
        transformed_vertices = vertices
    
    # Plot each edge
    for edge in edges:
        points = transformed_vertices[edge]
        ax.plot(points[:, 0], points[:, 1], points[:, 2], color="b")

# Animation function for rotating cube about the x-axis
def animate_rotation_x():
    fig = plt.figure()
    ax = fig.add_subplot(111, projection="3d")
    vertices, edges = cube_vertices_edges()

    def update(frame):
        ax.cla()
        ax.set_xlim([-1, 1])
        ax.set_ylim([-1, 1])
        ax.set_zlim([-1, 1])
        # Create a transformation matrix for rotation about x-axis
        transform = SE3.Rx(np.radians(frame))
        plot_cube(ax, vertices, edges, transform.A)

    ani = FuncAnimation(fig, update, frames=range(0, 360, 5), interval=50)
    plt.show()

# Animation function for rotating cube about all axes
def animate_rotation_all_axes():
    fig = plt.figure()
    ax = fig.add_subplot(111, projection="3d")
    vertices, edges = cube_vertices_edges()

    def update(frame):
        ax.cla()
        ax.set_xlim([-1, 1])
        ax.set_ylim([-1, 1])
        ax.set_zlim([-1, 1])
        # Create a composite transformation matrix rotating about all axes
        transform = SE3.Rx(np.radians(frame)) * SE3.Ry(np.radians(frame)) * SE3.Rz(np.radians(frame))
        plot_cube(ax, vertices, edges, transform.A)

    ani = FuncAnimation(fig, update, frames=range(0, 360, 5), interval=50)
    plt.show()

# Run the animations
animate_rotation_x()
animate_rotation_all_axes()
